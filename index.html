<script>
    // 🔗 แทนที่ลิงก์ด้านล่างด้วย Web App URL ที่ได้จาก Google Apps Script ของคุณ
    const WEB_APP_URL = 'https://docs.google.com/spreadsheets/d/1tREndRnvKsYMJtaNzpyj5SrEGu9vKQgG9aWo9vdy7qw/edit?usp=drivesdk';

    async function loadQueueData() {
        try {
            const response = await fetch(WEB_APP_URL);
            const data = await response.json();
            
            // แสดงผลภาพรวม
            document.getElementById('summary-box').innerText = `ความคืบหน้าภาพรวม: ${data.summary}`;

            const listContainer = document.getElementById('queue-list');
            listContainer.innerHTML = '';

            if (data.queues.length === 0) {
                listContainer.innerHTML = '<p style="text-align:center;">ยังไม่มีรายการคิวในขณะนี้</p>';
                return;
            }

            // วนลูปสร้างการ์ดแสดงผลทีละคิว
            data.queues.forEach((item, index) => {
                let badgeClass = 'pending';
                let statusText = item.status || 'กำลังดำเนินการ';
                
                if (item.status === 'เสร็จสิ้น') {
                    badgeClass = 'done';
                }

                const card = document.createElement('div');
                card.className = 'card';
                card.innerHTML = `
                    <div class="client-info">
                        <h3>#${index + 1} - ${item.name} (${item.platform || 'N/A'})</h3>
                        <p>แพ็กเกจ: ${item.type || '-'} | ขนาด: ${item.size || '-'}</p>
                        <p style="font-size: 0.8rem; color: #aaa; margin-top: 4px;">การชำระเงิน: ${item.payment || 'รอชำระ'}</p>
                    </div>
                    <div>
                        <span class="badge ${badgeClass}">${statusText}</span>
                    </div>
                `;
                listContainer.appendChild(card);
            });

        } catch (error) {
            console.error('Error fetching data:', error);
            document.getElementById('queue-list').innerHTML = '<p style="text-align:center; color: red;">ไม่สามารถโหลดข้อมูลได้ กรุณาตรวจสอบลิงก์ Web App อีกครั้ง</p>';
        }
    }

    loadQueueData();
</script>
